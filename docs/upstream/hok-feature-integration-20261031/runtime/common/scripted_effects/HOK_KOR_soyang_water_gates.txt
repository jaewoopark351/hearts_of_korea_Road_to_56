#20261031_kpopmodder: Check only the marked module and fixed Korean scope after peace, territorial changes, annexation or the daily fallback.
HOK_KOR_soyang_check_safety_effect = {
	if = {
		limit = { has_global_flag = HOK_KOR_soyang_managed_water }
		if = {
			limit = {
				NOT = {
					KOR = {
						exists = yes
						HOK_KOR_soyang_operation_conditions_trigger = yes
					}
				}
			}
			HOK_KOR_soyang_cleanup_effect = yes
		}
	}
}
