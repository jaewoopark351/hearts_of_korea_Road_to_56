#20261031_kpopmodder: Give a missing research specialization an eligible preferred core state, without replacing another facility.

#20261031_kpopmodder: Connect the existing land research focus to a facility and one independent scientist level.
HOK_KOR_establish_land_research_network = {
	if = {
		limit = { has_dlc = "Gotterdammerung" }
		custom_effect_tooltip = HOK_KOR_land_research_network_tt
		hidden_effect = {
			#20261031_kpopmodder: Use literal IDs after the target-version loader rejected this parameterized placement block.
			if = {
				limit = {
					NOT = { has_facility_specialization = specialization_land }
				}
				random_owned_controlled_state = {
					limit = {
						is_core_of = ROOT
						is_fully_controlled_by = ROOT
						can_construct_building = land_facility
						air_facility < 1
						land_facility < 1
						naval_facility < 1
						nuclear_facility < 1
					}
					prioritize = { 525 }
					construct_building_in_random_province = {
						land_facility = 1
					}
				}
			}
			if = {
				limit = {
					has_character = KOR_Yonggwan_Kim
					NOT = { has_country_flag = HOK_KOR_land_scientist_training_awarded }
				}
				KOR_Yonggwan_Kim = {
					add_scientist_level = { specialization = specialization_land level = 1 }
				}
				set_country_flag = HOK_KOR_land_scientist_training_awarded
			}
		}
	}
}
